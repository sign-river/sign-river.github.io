{{- $destination := .Destination -}} {{- $destinationLower := lower $destination -}} {{- $cleanDestination := replaceRE `[?#].*$` "" $destinationLower -}} {{- $extension := path.Ext $cleanDestination -}} {{- $downloadExtensions := slice ".7z" ".apk" ".bat" ".cmd" ".conf" ".csv" ".deb" ".dmg" ".doc" ".docx" ".exe" ".gz" ".img" ".ini" ".iso" ".json" ".msi" ".pdf" ".pkg" ".ps1" ".rar" ".rpm" ".sh" ".tar" ".tgz" ".toml" ".txt" ".xls" ".xlsx" ".yaml" ".yml" ".zip" -}} {{- $isBundledFile := or (strings.HasPrefix $cleanDestination "files/") (strings.HasPrefix $cleanDestination "/files/") (strings.Contains $cleanDestination "/files/") -}} {{- $isDownload := or $isBundledFile (in $downloadExtensions $extension) -}} {{- $isExternal := or (strings.HasPrefix $destinationLower "http://") (strings.HasPrefix $destinationLower "https://") (strings.HasPrefix $destinationLower "//") -}} {{- $linkKind := "internal" -}} {{- if $isDownload -}} {{- $linkKind = "download" -}} {{- else if $isExternal -}} {{- $linkKind = "external" -}} {{- end -}} {{- if $isDownload -}} {{ .Text | safeHTML }} {{- with $extension }} {{ strings.TrimPrefix "." . | upper }} {{- end }} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
